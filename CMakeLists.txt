# Bounded coupled power/thermal worker by Jason Stys. Targets/options: docs/CODE_INDEX.md.
cmake_minimum_required(VERSION 3.28)
project(SpacecraftPowerThermalLab VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(POWER_SANITIZE "Enable Clang ASan and UBSan" OFF)
option(POWER_COVERAGE "Enable GCC coverage" OFF)
if(POWER_SANITIZE AND MSVC)
  message(FATAL_ERROR "Combined sanitizer mode requires Linux Clang")
endif()
if(POWER_COVERAGE AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Coverage requires GCC; refusing an uninstrumented build")
endif()
add_library(power_model native/core.cpp)
target_include_directories(power_model PUBLIC native)
add_executable(power_core native/cli.cpp)
target_link_libraries(power_core PRIVATE power_model)
add_executable(core_tests native/tests.cpp)
target_link_libraries(core_tests PRIVATE power_model)
foreach(target power_model power_core core_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(POWER_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(POWER_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endforeach()
enable_testing()
add_test(NAME power_thermal_contracts COMMAND core_tests)
